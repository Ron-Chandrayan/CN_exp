client and server
import socket
s=socket.socket()
s.bind(("127.0.0.1",5000))
s.listen(1)
print("Waiting for client: ")

c,addr=s.accept()
print("Client connected",addr)

while True:
    msg=c.recv(1024).decode()
    if msg.lower() == "exit":
        c.send("Goodbye".encode())
        break
    print("Client: ",msg)
    reply=input("Server: ")
    c.send((reply).encode())

c.close()
s.close()
